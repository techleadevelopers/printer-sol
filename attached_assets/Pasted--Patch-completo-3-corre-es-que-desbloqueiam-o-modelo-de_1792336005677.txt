PATCH 1 — reserved release em todas as saídas (crítico)
Problema: execution.confirm vaza reserva em timeout, executionErr, e buy/sell com delta inválido. BuyNewTokenUSD só libera se !active — mas o estado já virou unknown (que É ativo). Após ~20 timeouts em um dia, bot trava com "insufficient free wallet balance" até restart.

Solução: liberar sempre no erro síncrono + defer no confirm. A proteção contra double-buy real fica em HasPurchaseForMint (DB) + leitura fresca de rpcs.Balance (live). Ambos são mais fortes que a reserva.

package positions

import (
	"fmt"
	"time"
)

type Limits struct {
	TakeProfitPct, StopLossPct, TrailingStopPct, MaxSlippagePct, MaxExposureSOL, MaxDailyLossSOL float64
	MaxHold                                                                                     time.Duration
}
type Position struct {
	Mint              string
	Quantity, CostSOL float64
	OpenedAt          time.Time
}

// ExitReason returns only deterministic limit triggers.
// currentSOL and peakSOL are mark prices converted to the entry denomination
// (e.g. cost * currentMcap/entryMcap). liquid=true only if the mark is
// verified; if false, exit is refused regardless of P&L.
func ExitReason(p Position, currentSOL, peakSOL float64, liquid bool, l Limits, now time.Time) string {
	if !liquid {
		return "insufficient verified liquidity"
	}
	if p.CostSOL <= 0 {
		return "cost pending reconciliation"
	}
	change := (currentSOL - p.CostSOL) / p.CostSOL * 100

	// 1. Hard floor
	if l.StopLossPct > 0 && change <= -l.StopLossPct {
		return "stop loss"
	}

	// 2. Ceiling
	if l.TakeProfitPct > 0 && change >= l.TakeProfitPct {
		return "take profit"
	}

	// 3. Trailing: only arms after peak >= +10% (configurable via TakeProfit
	//    implicitly — we hard-code 10 to avoid another knob; override with
	//    TrailingArmPct if you want it tunable).
	if l.TrailingStopPct > 0 && peakSOL > p.CostSOL {
		const armAt = 10.0
		peakChange := (peakSOL - p.CostSOL) / p.CostSOL * 100
		if peakChange >= armAt {
			drop := (peakSOL - currentSOL) / peakSOL * 100
			if drop >= l.TrailingStopPct {
				return fmt.Sprintf("trailing stop: peak %+.1f%%, drop %.1f%%", peakChange, drop)
			}
		}
	}

	// 4. Time stop
	if l.MaxHold > 0 && now.Sub(p.OpenedAt) >= l.MaxHold {
		return "max hold time"
	}
	return ""
}

func (m *Monitor) markTrade(ctx context.Context, mint string, marketCapSOL float64) {
	var err error
	if m.simulation {
		err = m.db.RecordPaperMark(mint, marketCapSOL)
	} else {
		err = m.db.RecordLiveMark(mint, marketCapSOL)

		returnPct, known, err = m.db.PaperReturnPct(mint)

			StopLossPct:     m.stopLoss,

			MaxHold:         m.maxHold,
		},

	)

		"mint": mint, "marketCapSol": marketCapSOL, "returnPct": returnPct,
		"peakMcapSol": peakMcap, "takeProfitPct": m.takeProfit,

PATCH 3 — Entrada < 250ms (async RugCheck + cache SOL/USD)
Problema: RugCheck HTTP (200–2000ms) + pricefeed HTTP (200–500ms) bloqueiam o caminho quente. Quando chega no Send, o primeiro pump já foi.

Solução: dividir Strategy.Evaluate em EvaluateFast (só DB) + Screen (rede), e disparar Screen em goroutine depois da entrada. Se o RugCheck reprovar, o trailing stop / SL já está armado. Perder 25% é melhor que perder 100% dos primeiros pumps.

internal/strategy/strategy.go
go
// EvaluateFast — caminho quente, sem rede.
func (s *Strategy) EvaluateFast(mint string, receivedAt time.Time) (bool, string) {
	if mint == "" {
		return false, "missing mint"
	}
	if s.maxAge > 0 && time.Since(receivedAt) > s.maxAge {
		return false, "token creation exceeded maximum entry age"
	}
	active, e := s.db.ActiveForMint(mint)
	if e != nil {
		return false, "database unavailable"
	}
	if active {
		return false, "existing unreconciled position"
	}
	return true, "fast-path accepted"
}

// Screen — chamada assíncrona pós-entrada.
func (s *Strategy) Screen(ctx context.Context, mint string) (string, error) {
	if s.screen == nil {
		return "no screen configured", nil
	}
	return s.screen(ctx, mint)
}

// Evaluate — mantida para compat; usa ambos.
func (s *Strategy) Evaluate(ctx context.Context, mint string, receivedAt time.Time) (bool, string) {
	ok, reason := s.EvaluateFast(mint, receivedAt)
	if !ok {
		return false, reason
	}
	if s.screen != nil {
		sr, err := s.screen(ctx, mint)
		if err != nil {
			return false, err.Error()
		}
		return true, sr
	}
	return true, reason
}

func (m *Monitor) consume(ctx context.Context) {
	for {
		select {
		case <-ctx.Done():
			return
		case e := <-m.q:
			latency := time.Since(e.Received).Milliseconds()
			if err := m.db.RecordEvent(e.Mint, e.Creator, e.Raw, e.Received, latency); err != nil {
				m.log.Warn("record event", zap.Error(err))
				m.bus.Publish("operational_error", map[string]string{"component": "database", "message": "unable to record token"})
			}
			ok, reason := m.strategy.EvaluateFast(e.Mint, e.Created) // SEM REDE
			m.log.Info("token evaluated (fast)", zap.String("mint", e.Mint), zap.Bool("eligible", ok), zap.String("reason", reason), zap.Int64("latency_ms", latency))
			if !ok || m.entry == nil {
				_ = m.db.RecordAssessment(e.Mint, false, reason)
				m.bus.Publish("token_rejected", map[string]any{"mint": e.Mint, "reason": reason, "latencyMs": latency})
				continue
			}

			// FIRE-AND-FORGET screen pós-entrada. Não bloqueia o caminho quente.
			go func(mint string) {
				sctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
				defer cancel()
				sreason, serr := m.strategy.Screen(sctx, mint)
				if serr != nil || !strings.HasPrefix(sreason, "RugCheck passed") {
					detail := sreason
					if serr != nil {
						detail = serr.Error()
					}
					m.log.Warn("post-entry risk flagged", zap.String("mint", mint), zap.String("reason", detail))
					_ = m.db.RecordAssessment(mint, false, "post-entry: "+detail)
					m.bus.Publish("risk_alert", map[string]any{
						"mint": mint, "reason": "post-entry risk screen", "detail": detail,
					})
				}
			}(e.Mint)

			id, entryErr := m.entry.BuyNewTokenUSD(ctx, e.Mint, m.tradeUSD, m.slippage, m.priorityFee, "pump", reason)
			if entryErr != nil {
				_ = m.db.RecordAssessment(e.Mint, false, entryErr.Error())
				m.bus.Publish("entry_rejected", map[string]any{"mint": e.Mint, "reason": entryErr.Error()})
				m.log.Info("approved token not entered", zap.String("mint", e.Mint), zap.Error(entryErr))
				continue
			}
			if assessmentErr := m.db.RecordAssessment(e.Mint, true, reason); assessmentErr != nil {
				m.log.Warn("record entered-token assessment", zap.Error(assessmentErr))
			}
			if m.simulation {
				_ = m.db.RecordPaperMark(e.Mint, e.MarketCapSOL)
			} else {
				_ = m.db.RecordLiveMark(e.Mint, e.MarketCapSOL)
			}
			m.tracked.Store(e.Mint, struct{}{})
			select {
			case m.subscriptions <- e.Mint:
			default:
				m.log.Warn("token trade subscription queue full", zap.String("mint", e.Mint))
			}
			m.bus.Publish("token_approved", map[string]any{"mint": e.Mint, "reason": reason, "latencyMs": latency})
			m.bus.Publish("entry_submitted", map[string]any{"mint": e.Mint, "tradeId": id, "simulation": m.simulation})
		}
	}
}

go
package pricefeed

import (
	"context"
	"encoding/json"
	"fmt"
	"io"
	"math"
	"net/http"
	"sync"
	"time"
)

type Provider interface {
	SOLUSD(context.Context) (float64, error)
}

type HTTP struct {
	url    string
	client *http.Client
	mu     sync.Mutex
	lastAt time.Time
	lastV  float64
	ttl    time.Duration
}

func New(url string) *HTTP {
	return &HTTP{url: url, client: &http.Client{Timeout: 5 * time.Second}, ttl: 30 * time.Second}
}

// SOLUSD caches for ttl — a stale-but-recent quote is dramatically safer
// than blocking a 250ms entry path on a 500ms HTTP round-trip.
func (h *HTTP) SOLUSD(ctx context.Context) (float64, error) {
	h.mu.Lock()
	if h.lastV > 0 && time.Since(h.lastAt) < h.ttl {
		v := h.lastV
		h.mu.Unlock()
		return v, nil
	}
	h.mu.Unlock()

	req, err := http.NewRequestWithContext(ctx, http.MethodGet, h.url, nil)
	if err != nil {
		return 0, err
	}
	res, err := h.client.Do(req)
	if err != nil {
		return 0, err
	}
	defer res.Body.Close()
	if res.StatusCode != http.StatusOK {
		return 0, fmt.Errorf("SOL/USD price HTTP %d", res.StatusCode)
	}
	var payload struct {
		Solana struct {
			USD float64 `json:"usd"`
		} `json:"solana"`
	}
	if err := json.NewDecoder(io.LimitReader(res.Body, 8192)).Decode(&payload); err != nil {
		return 0, err
	}
	if payload.Solana.USD <= 0 || math.IsNaN(payload.Solana.USD) || math.IsInf(payload.Solana.USD, 0) {
		return 0, fmt.Errorf("invalid SOL/USD price")
	}
	h.mu.Lock()
	h.lastV = payload.Solana.USD
	h.lastAt = time.Now()
	h.mu.Unlock()
	return payload.Solana.USD, nil
}
